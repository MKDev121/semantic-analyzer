// 75%: scalar types, operator precedence, and Boolean conditions.
int main() {
    int count = 2 + 3 * 4;
    float total = count + 0.5;
    char grade = 'A';
    bool ready = (count > 0) && (grade == 'A');

    if (ready) {
        total = total + 1.0;
    } else {
        total = 0.0;
    }
    while (count > 0) {
        count = count - 1;
    }
    return count;
}
