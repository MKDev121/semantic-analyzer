// Test 7: Semantic Errors - Non-boolean Conditions
int main() {
    int x = 5;
    float f = 1.0;

    if (x + 10) {   // Error: Condition in 'if' statement must evaluate to 'bool'
        x = x + 1;
    }

    while (f) {     // Error: Condition in 'while' statement must evaluate to 'bool'
        x = x - 1;
    }

    return 0;
}
