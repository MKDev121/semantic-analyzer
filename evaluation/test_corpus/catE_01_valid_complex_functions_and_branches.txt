// Category E - Test 01: Complex valid program with functions, branches, and loops
int maxVal(int a, int b) {
    int m;
    if (a > b) {
        m = a;
    } else {
        m = b;
    }
    return m;
}

int main() {
    int i = 0;
    int best = 0;
    while (i < 4) {
        best = maxVal(best, i * 3);
        i = i + 1;
    }
    return best;
}
