// Category B - Test 02: Nested if-else where all paths initialize x
int classify(int a, int b) {
    int x;
    if (a > 0) {
        if (b > 0) {
            x = 1;
        } else {
            x = 2;
        }
    } else {
        x = 3;
    }
    return x;
}

int main() {
    return classify(5, -1);
}
